3. ANIMATION SYSTEMS

Three animation approaches: CSS keyframes (simplest), SVG SMIL (no JS needed), Canvas requestAnimationFrame (most powerful). Mix as needed.

@keyframes swordswing {

}

← Pure SVG SMIL: rotating rune circle, pulsing orb, orbiting particles 
/* ===== SVG SMIL ANIMATION EXAMPLES ===== */

/* 1. Continuous rotation (magic circle, spin attack) */
<animateTransform attributeName="transform" type="rotate"
  from="0 CX CY" to="360 CX CY"
  dur="4s" repeatCount="indefinite"/>

/* 2. Pulsing scale / breathing */
<animateTransform attributeName="transform" type="scale"
  values="1;1.05;1" dur="3s" repeatCount="indefinite"
  additive="sum"/>

/* 3. Attribute animation (radius, opacity, color) */
<animate attributeName="r" values="10;15;10" dur="1.5s" repeatCount="indefinite"/>
<animate attributeName="opacity" values="1;0.5;1" dur="2s" repeatCount="indefinite"/>
<animate attributeName="fill" values="#ff0000;#ff8800;#ff0000" dur="0.5s" repeatCount="indefinite"/>

/* 4. Path motion (orbiting particle, patrol movement) */
<circle r="4" fill="#88aaff">
  <animateMotion dur="2s" repeatCount="indefinite"
    path="M0,0 m-25,0 a25,25 0 1,1 50,0 a25,25 0 1,1 -50,0"/>
</circle>

/* 5. Chained animations (windup then attack) */
<animateTransform id="windup" attributeName="transform" type="rotate"
  from="0" to="-40" dur="0.2s" begin="click" fill="freeze"/>
<animateTransform attributeName="transform" type="rotate"
  from="-40" to="60" dur="0.15s" begin="windup.end" fill="freeze"/>
<animateTransform attributeName="transform" type="rotate"
  from="60" to="0" dur="0.2s" begin="windup.end+0.15s" fill="freeze"/>

3c. Canvas requestAnimationFrame Sprite System

/* ===== CANVAS SPRITE ANIMATION SYSTEM ===== */

class Sprite {
  constructor(drawFn, x, y, scale=1) {
    this.draw = drawFn;   // function(ctx, state) — draws one frame
    this.x = x; this.y = y; this.scale = scale;
    this.state = 'idle';   // 'idle' | 'attack' | 'hurt' | 'dead'
    this.frame = 0;
    this.frameTimer = 0;
    this.FRAMES_PER_TICK = 4;  // change to adjust speed
  }

  update(dt) {
    this.frameTimer++;
    if (this.frameTimer >= this.FRAMES_PER_TICK) {
      this.frame++;
      this.frameTimer = 0;
    }
    // State machine
    if (this.state === 'attack' && this.frame > 8) {
      this.state = 'idle'; this.frame = 0;
    }
    if (this.state === 'hurt' && this.frame > 4) {
      this.state = 'idle'; this.frame = 0;
    }
  }

  render(ctx) {
    ctx.save();
    ctx.translate(this.x, this.y);
    ctx.scale(this.scale, this.scale);
    this.draw(ctx, { state: this.state, frame: this.frame });
    ctx.restore();
  }
}

/* Procedural draw function for a knight */
function drawKnight(ctx, { state, frame }) {
  const t = frame / 8;  // normalized time 0-1

  /* Idle breathing */
  const breathY = state === 'idle' ? Math.sin(frame * 0.3) * 1.5 : 0;

  /* Attack swing angle */
  let swordAngle = 0;
  if (state === 'attack') {
    if (frame < 3) swordAngle = -45 * (frame / 3);      // windup
    else if (frame < 6) swordAngle = -45 + 110 * ((frame-3)/3); // swing
    else swordAngle = 65 - 65 * ((frame-6)/2);          // return
  }

  /* Draw shadow */
  ctx.save();
  ctx.fillStyle = 'rgba(0,0,0,0.3)';
  ctx.scale(1, 0.3);
  ctx.beginPath();
  ctx.arc(0, 40, 18, 0, Math.PI*2);
  ctx.fill();
  ctx.restore();

  /* Draw body (Y shifts with breathe) */
  ctx.save();
  ctx.translate(0, breathY);

  /* Legs */
  ctx.fillStyle = '#8aaabb';
  ctx.fillRect(-12, 20, 9, 18);
  ctx.fillRect(3, 20, 9, 18);
  /* Boots */
  ctx.fillStyle = '#334455';
  ctx.fillRect(-13, 34, 11, 6);
  ctx.fillRect(2, 34, 11, 6);

  /* Body */
  const grad = ctx.createLinearGradient(-14,0,14,0);
  grad.addColorStop(0, '#c8d8e8');
  grad.addColorStop(0.5, '#8aaabb');
  grad.addColorStop(1, '#445566');
  ctx.fillStyle = grad;
  ctx.beginPath();
  ctx.roundRect(-14, -10, 28, 30, 3);
  ctx.fill();

  /* Sword arm (rotating for attack) */
  ctx.save();
  ctx.translate(18, -5);
  ctx.rotate(swordAngle * Math.PI / 180);
  ctx.fillStyle = '#ddeeff';
  ctx.fillRect(-2, -35, 4, 35);
  ctx.fillStyle = '#cc9900';
  ctx.fillRect(-6, -3, 12, 3);  // crossguard
  ctx.restore();

  /* Head / Helmet */
  const hgrad = ctx.createLinearGradient(-12,-30,12,-30);
  hgrad.addColorStop(0, '#c8d8e8');
  hgrad.addColorStop(1, '#445566');
  ctx.fillStyle = hgrad;
  ctx.beginPath();
  ctx.ellipse(0, -20, 12, 13, 0, 0, Math.PI*2);
  ctx.fill();

  /* Visor */
  ctx.fillStyle = '#334455';
  ctx.beginPath();
  ctx.ellipse(0, -16, 10, 6, 0, 0, Math.PI);
  ctx.fill();

  ctx.restore();
}

/* Game loop */
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');
const knight = new Sprite(drawKnight, 100, 100, 1.5);

function loop(timestamp) {
  ctx.clearRect(0, 0, canvas.width, canvas.height);
  knight.update(timestamp);
  knight.render(ctx);
  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);

/* Trigger attack on click */
canvas.addEventListener('click', () => {
  knight.state = 'attack';
  knight.frame = 0;
});
